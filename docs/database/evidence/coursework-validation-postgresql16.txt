  id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  name text NOT NULL,
  email text NOT NULL UNIQUE,
  password_hash text NOT NULL,
  whatsapp text,
  active boolean NOT NULL DEFAULT true,

-- Catálogo de permissões que podem ser associadas aos papéis.

);

-- Solicitações dos usuários; código, estado e campos obrigatórios são validados no banco.

  id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,

  service text NOT NULL,

  urgency text NOT NULL CHECK (urgency IN ('LOW','MEDIUM','HIGH')),

    CHECK (status IN ('Pendente','Aceita para atendimento','Em atendimento','Concluída','Recusada','Cancelada')),

  review_notes text,

  CONSTRAINT demands_completion_check
    CHECK (status <> 'Concluída' OR nullif(btrim(completion_note), '') IS NOT NULL)

-- Impede transições fora do fluxo definido e mantém a data de alteração.
CREATE FUNCTION transition_guard()
RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
  IF NEW.status <> OLD.status AND NOT (
    (OLD.status = 'Pendente' AND NEW.status IN ('Aceita para atendimento','Recusada','Cancelada')) OR
    (OLD.status = 'Aceita para atendimento' AND NEW.status IN ('Em atendimento','Cancelada')) OR
    (OLD.status = 'Em atendimento' AND NEW.status IN ('Concluída','Cancelada'))
  ) THEN
    RAISE EXCEPTION 'transição de demanda inválida: % -> %', OLD.status, NEW.status;
  END IF;
  NEW.updated_at = now();
  RETURN NEW;
END
$$;
CREATE FUNCTION
-- Executa a validação da transição antes de gravar a nova situação.
CREATE TRIGGER transition_guard
BEFORE UPDATE OF status ON demands
FOR EACH ROW EXECUTE FUNCTION transition_guard();

-- Registra publicação e retirada no mesmo commit da alteração da página.
CREATE FUNCTION page_publish_audit()
RETURNS trigger
LANGUAGE plpgsql
AS $$
DECLARE
  actor_id bigint;
BEGIN
  IF NEW.published_version_id IS DISTINCT FROM OLD.published_version_id THEN
    actor_id := nullif(current_setting('ites.actor_id', true), '')::bigint;
    IF actor_id IS NULL THEN
      RAISE EXCEPTION 'ator obrigatório para publicar ou retirar uma página';
    END IF;
    INSERT INTO page_events(page_id, version_id, actor_id, action)
    VALUES (
      NEW.id,
      NEW.published_version_id,
      actor_id,
      CASE WHEN NEW.published_version_id IS NULL THEN 'UNPUBLISH' ELSE 'PUBLISH' END
    );
  END IF;
  RETURN NEW;
END
$$;
CREATE FUNCTION
-- Registra publicação e retirada na mesma transação da alteração da página.
CREATE TRIGGER page_publish_audit
AFTER UPDATE OF published_version_id ON pages
FOR EACH ROW EXECUTE FUNCTION page_publish_audit();

-- Procedimentos agrupam gravações relacionadas em uma única chamada transacional.
-- Aplica uma decisão e seu evento de auditoria na mesma transação.
CREATE PROCEDURE transition_demand(
  p_demand_id bigint,
  p_actor_id bigint,
  p_status text,
  p_justification text,
  p_completion_note text DEFAULT NULL
)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
DECLARE
  previous_status text;
BEGIN
  IF nullif(btrim(p_justification), '') IS NULL THEN
    RAISE EXCEPTION 'justificativa obrigatória';
  END IF;
  SELECT status INTO previous_status FROM demands WHERE id = p_demand_id FOR UPDATE;
  IF NOT FOUND THEN
    RAISE EXCEPTION 'demanda não encontrada';
  END IF;
  IF p_status = previous_status THEN
    RAISE EXCEPTION 'a demanda já está nessa situação';
  END IF;
  UPDATE demands
  SET status = p_status,
      completion_note = CASE WHEN p_status = 'Concluída' THEN p_completion_note ELSE completion_note END
  WHERE id = p_demand_id;
  INSERT INTO demand_events(demand_id, actor_id, old_status, new_status, justification)
  VALUES (p_demand_id, p_actor_id, previous_status, p_status, btrim(p_justification));
END
$$;
CREATE PROCEDURE
-- Publica somente uma revisão pertencente à página e audita por gatilho.
CREATE PROCEDURE publish_page(p_page_id bigint, p_version_id bigint, p_actor_id bigint)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM page_versions WHERE id = p_version_id AND page_id = p_page_id
  ) THEN
    RAISE EXCEPTION 'versão não pertence à página informada';
  END IF;
  PERFORM set_config('ites.actor_id', p_actor_id::text, true);
  UPDATE pages SET published_version_id = p_version_id WHERE id = p_page_id;
END
$$;
CREATE PROCEDURE
-- Retira uma publicação e conserva a versão no histórico de auditoria.
CREATE PROCEDURE unpublish_page(p_page_id bigint, p_actor_id bigint)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pages WHERE id = p_page_id) THEN
    RAISE EXCEPTION 'página não encontrada';
  END IF;
  PERFORM set_config('ites.actor_id', p_actor_id::text, true);
  UPDATE pages SET published_version_id = NULL WHERE id = p_page_id;
END
$$;
CREATE PROCEDURE
-- Concede um papel e registra somente concessões efetivamente novas.
CREATE PROCEDURE assign_role(p_user_id bigint, p_role_id bigint, p_actor_id bigint)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
DECLARE
  affected_rows integer;
BEGIN
  INSERT INTO user_roles(user_id, role_id)
  VALUES (p_user_id, p_role_id)
  ON CONFLICT DO NOTHING;
  GET DIAGNOSTICS affected_rows = ROW_COUNT;
  IF affected_rows > 0 THEN
    INSERT INTO grant_audit(actor_id, subject_user_id, role_id, action)
    VALUES (p_actor_id, p_user_id, p_role_id, 'GRANT');
  END IF;
END
$$;
CREATE PROCEDURE
-- Revoga um papel e registra somente revogações efetivamente aplicadas.
CREATE PROCEDURE revoke_role(p_user_id bigint, p_role_id bigint, p_actor_id bigint)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
DECLARE
  affected_rows integer;
BEGIN
  DELETE FROM user_roles WHERE user_id = p_user_id AND role_id = p_role_id;
  GET DIAGNOSTICS affected_rows = ROW_COUNT;
  IF affected_rows > 0 THEN
    INSERT INTO grant_audit(actor_id, subject_user_id, role_id, action)
    VALUES (p_actor_id, p_user_id, p_role_id, 'REVOKE');
  END IF;
END
$$;

-- Verificações reexecutáveis da carga acadêmica, das restrições e das rotinas.
-- Uma exceção ou resultado diferente do esperado encerra o script com erro.
DO $$
DECLARE
  table_name text;
  row_total bigint;
  table_names text[] := ARRAY[
    'users','user_roles','role_permissions','organizations','demands','demand_events',
    'pages','page_versions','page_events','grant_audit'
  ];
BEGIN
  -- Confere a quantidade mínima de registros nas tabelas de negócio e associação.
  FOREACH table_name IN ARRAY table_names LOOP
    EXECUTE format('SELECT count(*) FROM %I', table_name) INTO row_total;
    IF row_total < 20 THEN
      RAISE EXCEPTION 'a tabela % contém apenas % linhas', table_name, row_total;
    END IF;
  END LOOP;

  -- Os catálogos mantêm somente papéis e permissões com significado funcional.
  IF (SELECT count(*) FROM roles) <> 4 THEN
    RAISE EXCEPTION 'o catálogo deve conter os quatro papéis de negócio definidos';
  END IF;
  IF (SELECT count(*) FROM permissions) <> 13 THEN
    RAISE EXCEPTION 'o catálogo deve conter as treze permissões de negócio definidas';
  END IF;
  -- Confere valores positivos, inexistentes e contagens esperadas das funções.
  IF demand_count('Pendente') < 1 OR published_page_count() <> 20 OR
     requester_demand_count((SELECT id FROM users WHERE email = 'pessoa01@example.invalid')) < 1 OR
     requester_demand_count(999999) <> 0 OR demand_count('SITUACAO_INEXISTENTE') <> 0 THEN
    RAISE EXCEPTION 'indicadores retornaram valores incompatíveis com a carga';
  END IF;

  -- Testa decisão, avanço de atendimento e conclusão com justificativa.
  DECLARE
    pending_id bigint;
    v_actor_id bigint;
    v_page_id bigint;
    v_version_id bigint;
    rejected boolean;
    event_total bigint;
    audit_before bigint;
    subject_id bigint;
    requester_role_id bigint;
  BEGIN
    SELECT id INTO v_actor_id FROM users WHERE email = 'pessoa26@example.invalid';
    SELECT id INTO pending_id FROM demands WHERE status = 'Pendente' ORDER BY id LIMIT 1;
    CALL transition_demand(pending_id, v_actor_id, 'Aceita para atendimento', 'Demanda aceita no teste positivo');
    CALL transition_demand(pending_id, v_actor_id, 'Em atendimento', 'Atendimento iniciado no teste positivo');

    rejected := false;
    BEGIN
      CALL transition_demand(pending_id, v_actor_id, 'Concluída', 'Conclusão sem nota', NULL);
    EXCEPTION WHEN check_violation THEN
      rejected := true;
    END;
    IF NOT rejected THEN
      RAISE EXCEPTION 'a conclusão sem nota deveria ser recusada';
    END IF;

    CALL transition_demand(pending_id, v_actor_id, 'Concluída', 'Atendimento finalizado', 'Atividade realizada e registrada');
    rejected := false;
    BEGIN
      CALL transition_demand(pending_id, v_actor_id, 'Pendente', 'Tentativa de retrocesso', NULL);
    EXCEPTION WHEN OTHERS THEN
      rejected := true;
    END;
    IF NOT rejected THEN
      RAISE EXCEPTION 'uma transição terminal inválida deveria ser recusada';
    END IF;

    rejected := false;
    BEGIN
      CALL transition_demand(
        (SELECT id FROM demands WHERE status = 'Pendente' ORDER BY id LIMIT 1),
        v_actor_id, 'Aceita para atendimento', '   ', NULL
      );
    EXCEPTION WHEN OTHERS THEN
      rejected := true;
    END;
    IF NOT rejected THEN
      RAISE EXCEPTION 'uma decisão sem justificativa deveria ser recusada';
    END IF;

    SELECT count(*) INTO event_total FROM demand_events WHERE demand_id = pending_id;
    IF event_total <> 3 THEN
      RAISE EXCEPTION 'eram esperados três eventos para a demanda de validação';
    END IF;

    rejected := false;
    BEGIN
      UPDATE demand_events SET justification = 'alterado'
      WHERE id = (SELECT id FROM demand_events WHERE demand_id = pending_id ORDER BY id LIMIT 1);
    EXCEPTION WHEN OTHERS THEN
      rejected := true;
    END;
    IF NOT rejected THEN
      RAISE EXCEPTION 'o gatilho deveria impedir alteração do histórico';
    END IF;

    SELECT id INTO v_page_id FROM pages ORDER BY id LIMIT 1;
    INSERT INTO page_versions(page_id, revision, document, author_id)
    SELECT v_page_id, 2, pv.document || jsonb_build_object('revisaoTeste', true), v_actor_id
    FROM page_versions pv WHERE pv.page_id = v_page_id AND pv.revision = 1;
    SELECT pv.id INTO v_version_id FROM page_versions pv WHERE pv.page_id = v_page_id AND pv.revision = 2;
    CALL publish_page(v_page_id, v_version_id, v_actor_id);
    CALL unpublish_page(v_page_id, v_actor_id);
    IF (SELECT count(*) FROM page_events pe WHERE pe.page_id = v_page_id) < 3 THEN
      RAISE EXCEPTION 'publicação e retirada não foram registradas pelo gatilho';
    END IF;
    IF published_page_count() <> 19 THEN
      RAISE EXCEPTION 'retirada de publicação não atualizou o indicador';
    END IF;

    rejected := false;
    BEGIN
      CALL publish_page(v_page_id, v_version_id + 100000, v_actor_id);
    EXCEPTION WHEN OTHERS THEN
      rejected := true;
    END;
    IF NOT rejected THEN
      RAISE EXCEPTION 'a publicação de versão inexistente deveria ser recusada';
    END IF;

    SELECT id INTO subject_id FROM users WHERE email = 'pessoa30@example.invalid';
    SELECT id INTO requester_role_id FROM roles WHERE code = 'REQUESTER';
    SELECT count(*) INTO audit_before FROM grant_audit
      WHERE subject_user_id = subject_id AND role_id = requester_role_id;
    CALL assign_role(subject_id, requester_role_id, v_actor_id);
    IF (SELECT count(*) FROM grant_audit WHERE subject_user_id = subject_id AND role_id = requester_role_id) <> audit_before THEN
      RAISE EXCEPTION 'concessão duplicada não deveria gerar auditoria';
    END IF;
    CALL revoke_role(subject_id, requester_role_id, v_actor_id);
    CALL assign_role(subject_id, requester_role_id, v_actor_id);
    IF NOT EXISTS (SELECT 1 FROM user_roles WHERE user_id = subject_id AND role_id = requester_role_id) OR
       (SELECT count(*) FROM grant_audit WHERE subject_user_id = subject_id AND role_id = requester_role_id) <> audit_before + 2 THEN
      RAISE EXCEPTION 'concessão e revogação deveriam alterar papel e trilha de auditoria';
    END IF;

    rejected := false;
    BEGIN
      INSERT INTO demands(requester_id, capacity, service, description, urgency)
      VALUES (v_actor_id, 'OTHER', 'formação', 'Integridade de condição', 'LOW');
    EXCEPTION WHEN check_violation THEN
      rejected := true;
    END;
    IF NOT rejected THEN
      RAISE EXCEPTION 'a condição OTHER sem descrição deveria ser recusada';
    END IF;
  END;
END
$$;

-- Execute como proprietário do banco em um ambiente acadêmico isolado.
-- Contas LOGIN são provisionadas pelo administrador fora do repositório.
-- Cria papéis de grupo sem credenciais para separar responsabilidades técnicas.
DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'ites_runtime') THEN
    CREATE ROLE ites_runtime NOLOGIN;
  END IF;
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'ites_reporting') THEN
    CREATE ROLE ites_reporting NOLOGIN;
  END IF;
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'ites_migration') THEN
    CREATE ROLE ites_migration NOLOGIN;
  END IF;
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'ites_database_admin') THEN
    CREATE ROLE ites_database_admin NOLOGIN;
  END IF;
END
$$;
DO
